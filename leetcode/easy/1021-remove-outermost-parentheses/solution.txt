class Solution {
    public String removeOuterParentheses(String s) {
        StringBuilder sb = new StringBuilder();
        int counter = 0;

        for (int i = 0; i < s.length(); i ++) {
            char ch = s.charAt(i);
            if (ch == '(') {
                if (counter > 0) {
                    sb.append(ch);
                }
                counter ++;
            } else {
                if (counter > 1) {
                    sb.append(ch);
                }
                counter --;
            }
        }
        return sb.toString();
    }
}
