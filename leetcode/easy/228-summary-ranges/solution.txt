class Solution {
    public List<String> summaryRanges(int[] nums) {
        List<String> result = new ArrayList<>();
        if (nums.length == 0) {
            return result;
        }
        
        int l = 0;
        while (l < nums.length) {
            int start = nums[l];
            int r = l;
            while (r + 1 < nums.length && nums[r + 1] == nums[r] + 1) {
                r++;
            }
            
            if (nums[r] == start) {
                result.add(String.valueOf(start));
            } else {
                result.add(start + "->" + nums[r]);
            }
            
            l = r + 1;
        }
        return result;
    }
}
