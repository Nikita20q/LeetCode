class Solution {
    public double findMaxAverage(int[] nums, int k) {
        int maxSum = 0;
        for (int i = 0; i < k; i ++) {
            maxSum += nums[i];
        }
        int sum = 0;
        int l = 0;
        for (int r = 0; r < nums.length; r ++) {
            sum += nums[r];
            if (r - l + 1 > k) {
                sum -= nums[l];
                l++;
            } 
            if (r - l + 1 == k) maxSum = Math.max(maxSum, sum);
        }
        return (double)maxSum / (double)(k);
    }
}
