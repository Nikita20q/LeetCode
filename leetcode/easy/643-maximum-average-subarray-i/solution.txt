class Solution {
    public double findMaxAverage(int[] nums, int k) {
        int maxSum = 0;
        for (int i = 0; i < k; i ++) {
            maxSum += nums[i];
        }
        int sum = maxSum;
        int l = 0;
        for (int r = k; r < nums.length; r ++, l++) {
            sum += nums[r];
            sum -= nums[l];
            maxSum = Math.max(maxSum, sum);
        }
        return maxSum / (double)(k);
    }
}
