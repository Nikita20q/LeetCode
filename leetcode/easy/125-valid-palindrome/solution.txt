class Solution {
    public boolean isPalindrome(String s) {
        int r = s.length() - 1;
        int l = 0;
        while (l < r) {
            while (r >= 0 && (!Character.isLetter(s.charAt(r)) && !Character.isDigit(s.charAt(r)))) r --;
            while (l < s.length() && (!Character.isLetter(s.charAt(l)) && !Character.isDigit(s.charAt(l)))) l ++;
            if (r >= 0 && l < s.length() && Character.toLowerCase(s.charAt(r)) != Character.toLowerCase(s.charAt(l))) return false;
            l++;
            r--;
        }
        return true;
    }
}
