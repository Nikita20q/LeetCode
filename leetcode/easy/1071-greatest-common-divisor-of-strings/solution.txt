    // Проверяет, состоит ли s из повторений pattern
    private boolean isRepeated(String s, String pattern) {
        int pLen = pattern.length();
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) != pattern.charAt(i % pLen)) {
                return false;
            }
        }
        return true;
    }
