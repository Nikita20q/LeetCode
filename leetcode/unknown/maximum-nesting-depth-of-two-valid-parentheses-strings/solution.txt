class Solution {
    public int[] maxDepthAfterSplit(String seq) {
        int n = seq.length();
        int[] arr = new int[n];
        for (int i = 0; i < n; i ++) {
            if (seq.charAt(i) == '(') arr[i] = i % 2;
            else arr[i] = ((i + 1) % 2);
        }
        return arr;
    }
}
