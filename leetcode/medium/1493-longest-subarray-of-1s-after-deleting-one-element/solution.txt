class Solution {
    public int longestSubarray(int[] nums) {
        int l = 0;
        int zeroIdx = -1;
        int zeroCount = 0;
        int res = 0;
        for (int r = 0; r < nums.length; r ++) {
            if (nums[r] == 0 && zeroCount == 0) {
                zeroIdx = r;
                zeroCount = 1;
            } else if (nums[r] == 0 && zeroCount == 1) {
                l = zeroIdx + 1;
                zeroIdx = r;
            }
            res = Math.max(res, r - l);
        }
        return res;
    }
}
