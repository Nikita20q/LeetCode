// [1, 1, 0, 0, 1, 1]

class Solution {
    public int longestSubarray(int[] nums) {
        int l = 0;
        int res = 0;
        int lastZero = -1;

        for (int r = l; r < nums.length; r ++) {
            if (nums[r] == 0) {
                l = lastZero + 1;
                lastZero = r;
            }
            res = Math.max(res, r - l);
        }

        return res;
    }
}
