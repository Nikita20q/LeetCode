// 24 24 12 4
class Solution {
    public int[] productExceptSelf(int[] nums) {
        int[] res = new int[nums.length];
        res[0] = nums[0];
        for (int i = 1; i < nums.length; i ++) {
            res[i] = res[i - 1] * nums[i];
        }

        int sufix = 1;
        for (int i = nums.length - 1; i >= 0; i --) {
            if (i > 0) {
                res[i] = res[i - 1] * sufix;
            } else {
                res[i] = sufix;
            }
            sufix *= nums[i];
        }
        return res;
    }
}
