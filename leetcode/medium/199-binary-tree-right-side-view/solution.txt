y tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<Integer> rightSideView(TreeNode root) {
        List<Integer> res = new ArrayList<>();
        recFunc(root, res, 0);
        return res;
    }

    public void recFunc(TreeNode node, List<Integer> res, int depth) {
        if (node == null) return;
        if (depth == res.size()) res.add(node.val);
        recFunc(node.right, res, depth + 1);
        recFunc(node.left, res, depth + 1);
    }
}
