class Solution {
    public int[] searchRange(int[] nums, int target) {
        if (nums.length == 0) return new int[]{-1, -1};
        
        int l = 0;
        int r = nums.length - 1;
        int first = -1;

            int mid = l + (r - l) / 2;

                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }

            first = l;
