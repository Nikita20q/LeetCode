class Solution {
    public int maxVowels(String s, int k) {
        Set<Character> vof = Set.of('a', 'e', 'i', 'o', 'u', 'A', 'E', 'I', 'O', 'U');
        int counter = 0;
        for (int i = 0; i < k; i ++) {
            if (vof.contains(s.charAt(i))) counter ++;
        }
        int maxCounter = counter;
        for (int r = k, l = 0; r < s.length(); r ++, l ++) {
            if (vof.contains(s.charAt(r))) counter ++;
            if (vof.contains(s.charAt(l))) counter --;
            maxCounter = Math.max(maxCounter, counter);
        }
        return maxCounter;
    }
}
